Choose env

Pick the env by where a process can run, then select it per conversation

  1. Files and commands on this machine: new NodeExecutionEnv({ cwd }).
  2. Elsewhere, and a process can run there with stdin and stdout piped (ssh, docker exec -i, kubectl exec -i): new RemoteExecutionEnv({ connection, id, cwd }) with sshConnection(target).connection or new Connection({ command }). For a container, copy packagedDaemon({ platform, arch }) in first, as src/envs/remote.ts does.
  3. No process can run there: implement ExecutionEnv. Return Results, check the abort signal first, and set id per namespace. Prove it with createEnvConformance. src/envs/memory.ts is a complete example.
  4. In every case, env(target) reads the conversation’s choice from a document through target.read and returns a cached env per (choice, cwd), as src/envs/index.ts does.

Resources

Load AGENTS.md and skills through input.env; tools open them through api.env

  1. Before each request, the generation task calls HarnessOptions.env and hands the result to every section as input.env.
  2. The sections ask the resource loader (src/resources/extensions.ts). The loader (src/resources/loader.ts) starts a watch, then lists and reads through that env.
  3. Durable appends only the sections whose text changed, as a pi.system entry.
  4. During the work, the model calls the built-in read tool on a skill’s location from the catalog.
  5. The read tool reads through api.env, the same environment.

export type HarnessParts = {
	readonly storage: Storage;
	/** pi-ai: the model catalog and its credentials. */
	readonly models: Models;
	readonly envs: Environments;
	readonly settings?: HarnessSettings;
	readonly onReport?: (error: unknown) => void;

export type OpenHarness = {
	readonly harness: Harness;
	readonly registry: Registry;
	/** Close the Harness, then what it used: resource watchers and environments. */
	close(context: Context): Promise<void>;
};

/**
 * The composition root. pi-ai supplies models, pi-durable runs conversations durably on `storage`, and the
 * environments (local, pi-env remotes, memory) decide where tools run and where the prompt's resources come from.
 * Nothing here depends on which environment a conversation selects.
 */
export async function openHarness(parts: HarnessParts, context: Context): Promise<OpenHarness> {
	const loader = new ResourceLoader();
	const registry = createRegistry();
	for (const extension of [CodingTools, Preamble, createContextFiles(loader), createSkills(loader)]) {
		registry.install(extension);
	}
	const harness = await Harness.open(
		parts.storage,
		{ models: parts.models, registry, env: parts.envs.env, settings: parts.settings, onReport: parts.onReport },
		context,
	);
	return {
		harness,
		registry,
		async close(closeContext) {
			await harness.close(closeContext);
			await loader.close(closeContext);
			await parts.envs.cleanup(closeContext);
		},
	};
}

import type { Context } from "@earendil-works/chord";
import { defineDoc, type EnvTarget } from "@earendil-works/pi-durable";
import type { ExecutionEnv } from "@earendil-works/pi-durable/env";
import { NodeExecutionEnv } from "@earendil-works/pi-durable/env/node";
import { type Connection, RemoteExecutionEnv } from "@earendil-works/pi-env";
import { MemoryExecutionEnv, type MemoryVolume } from "./memory.ts";
import { dockerTransport, sshTransport } from "./remote.ts";

/**
 * Where a conversation's tools run, as a URI:
 * - `local`: this machine (`NodeExecutionEnv`)
 * - `ssh://[user@]host[:port]`: a host reached over SSH (pi-env)
 * - `docker://<container>`: a running container (pi-env)
 * - `memory://<volume>`: an in-memory volume the host registered (`MemoryExecutionEnv`)
 *
 * Stored per conversation, so a restart, or a different worker, rebuilds the same environment. `fork: "current"`: a
 * fork keeps running where its parent runs.
 */
export const EnvDoc = defineDoc<{ uri: string }>({
	kind: "harness.env",
	version: 1,
	scope: "conversation",
	history: "latest",
	fork: "current",
	initial: () => ({ uri: "local" }),
});

/**
 * Builds and caches execution environments. `env` is the Harness's `HarnessOptions.env`: Durable calls it for every
 * tool call (`api.env`) and before every request (`input.env` of the prompt sections), with the conversation's ID, its
 * agent `cwd`, and committed reads, so the environment follows the conversation's `EnvDoc` and `cwd` without a restart.
 */
export class Environments {
	/** Volumes `memory://<name>` URIs resolve to. */
	readonly volumes = new Map<string, MemoryVolume>();
	readonly #defaultUri: string;
	readonly #localCwd: string;
	readonly #envs = new Map<string, Promise<ExecutionEnv>>();
	/** One daemon connection per remote, shared by its working directories. */
	readonly #connections = new Map<string, Connection>();

	constructor(options: { defaultUri?: string; localCwd?: string } = {}) {
		this.#defaultUri = options.defaultUri ?? "local";
		this.#localCwd = options.localCwd ?? process.cwd();
	}

	readonly env = async ({ conversationId, cwd, read }: EnvTarget, context: Context): Promise<ExecutionEnv> => {
		const selected = await read.snapshot(EnvDoc, conversationId, context);
		return this.get(selected?.uri ?? this.#defaultUri, cwd);
	};

	/** The environment for `uri` in `cwd`; without `cwd`, its default directory. */
	get(uri: string, cwd?: string): Promise<ExecutionEnv> {
		const key = `${uri}\0${cwd ?? ""}`;
		let env = this.#envs.get(key);
		if (env === undefined) {
			env = this.#create(uri, cwd);
			this.#envs.set(key, env);
			// A failed build, such as an unreachable host, is retried at the next use.
			env.catch(() => this.#envs.delete(key));
		}
		return env;
	}

	/** Kill what the environments still run and close remote connections; for the host's shutdown. */
	async cleanup(context: Context): Promise<void> {
		const envs = await Promise.allSettled(this.#envs.values());
		this.#envs.clear();
		for (const env of envs) if (env.status === "fulfilled") await env.value.cleanup(context);
		for (const connection of this.#connections.values()) connection.close();
		this.#connections.clear();
	}

	async #create(uri: string, cwd: string | undefined): Promise<ExecutionEnv> {
		if (uri === "local") return new NodeExecutionEnv({ cwd: cwd ?? this.#localCwd });
		const url = new URL(uri);
		if (url.protocol === "memory:") {
			const volume = this.volumes.get(url.hostname);
			if (volume === undefined) throw new Error(`No memory volume "${url.hostname}"`);
			return new MemoryExecutionEnv(volume, cwd ?? "/");
		}
		let connection = this.#connections.get(uri);
		if (connection === undefined) {
			connection = remoteTransport(url);
			this.#connections.set(uri, connection);
		}
		// Equal ids promise the same files at the same paths: one id per remote, whatever the directory.
		return new RemoteExecutionEnv({ connection, id: `pi-env:${uri}`, cwd: cwd ?? (await connection.info()).cwd });
	}
}

function remoteTransport(url: URL): Connection {
	switch (url.protocol) {
		case "ssh:":
			return sshTransport({
				host: url.hostname,
				...(url.username === "" ? {} : { user: decodeURIComponent(url.username) }),
				...(url.port === "" ? {} : { port: Number(url.port) }),
			});
		case "docker:":
			return dockerTransport(url.hostname);
		default:
			throw new Error(`Unsupported environment "${url.href}"`);
	}
}

import { execFile } from "node:child_process";
import { createHash } from "node:crypto";
import { readFile } from "node:fs/promises";
import { homedir } from "node:os";
import { join } from "node:path";
import { promisify } from "node:util";
import { Connection, packagedDaemon, sshConnection } from "@earendil-works/pi-env";

const run = promisify(execFile);
const ARCH: Record<string, "x64" | "arm64"> = { x86_64: "x64", amd64: "x64", aarch64: "arm64", arm64: "arm64" };

/**
 * pi-env over SSH: detects the host, deploys the daemon (verified by its SHA-256) and starts it on the first operation.
 * Host keys are checked strictly against `~/.ssh/known_hosts` under the host's own name, so hosts the user already
 * trusts work and unknown ones fail with `HostKeyUnknownError`.
 */
export function sshTransport(target: { host: string; user?: string; port?: number }): Connection {
	const knownHostsFile = join(homedir(), ".ssh", "known_hosts");
	return sshConnection({ ...target, knownHostsFile, hostKeyAlias: target.host }).connection;
}

/**
 * pi-env inside a running container: copies the packaged Linux daemon in under its content hash once, then starts it
 * with `docker exec -i`. The daemon's protocol only needs stdin and stdout, so any command that pipes them to a
 * process somewhere (`kubectl exec -i`, `fly ssh console`, ...) is a transport the same way.
 */
export function dockerTransport(container: string): Connection {
	return new Connection({
		command: async () => {
			const machine = (await run("docker", ["exec", container, "uname", "-m"])).stdout.trim();
			const arch = ARCH[machine];
			if (arch === undefined) throw new Error(`No pi-env daemon for ${machine} containers`);
			const binary = packagedDaemon({ platform: "linux", arch });
			const hash = createHash("sha256")
				.update(await readFile(binary))
				.digest("hex");
			const path = `/tmp/pi-env-${hash.slice(0, 16)}`;
			const present = await run("docker", ["exec", container, "test", "-x", path]).then(
				() => true,
				() => false,
			);
			if (!present) await run("docker", ["cp", binary, `${container}:${path}`]);
			return ["docker", "exec", "-i", container, path];
		},
	});
}

/** One in-memory disk. Every `MemoryExecutionEnv` over it shares its `id` and sees the same files. */
export class MemoryVolume {
	readonly id: string;
	readonly nodes = new Map<string, Node>([["/", { kind: "directory", mtimeMs: Date.now() }]]);
	readonly listeners = new Set<(path: string) => void>();

	constructor(name: string) {
		this.id = `memory:${name}`;

	changed(...paths: string[]): void {
		for (const listener of this.listeners) for (const path of paths) listener(path);

/**
 * A custom `ExecutionEnv` without a host: files live in a `MemoryVolume`, and there is no shell, so `exec` fails
 * with `shell_unavailable` and the `bash` tool answers with an error result. It runs wherever JavaScript runs, which
 * makes it a sandbox, a test fixture, and the template for environments backed by something else (a KV store, a
 * browser).
 */
export class MemoryExecutionEnv implements ExecutionEnv {
	readonly volume: MemoryVolume;
	cwd: string;

	constructor(volume: MemoryVolume, cwd = "/") {
		this.volume = volume;
		this.cwd = cwd;
	}

	get id(): string {
		return this.volume.id;
	}

	async absolutePath(path: string, context: Context) {
		return this.#run(context, () => this.#resolve(path));
	}

	async joinPath(parts: string[], context: Context) {
		return this.#run(context, () => posix.join(...parts));
	}

	async readBinaryFile(path: string, context: Context) {
		return this.#run(context, () => this.#file(this.#resolve(path)).data.slice());
	}

	async readTextFile(path: string, context: Context) {
		return this.#run(context, () => decode(this.#file(this.#resolve(path)).data));
	}

	async readTextLines(path: string, options: { maxLines?: number } | undefined, context: Context) {
		return this.#run(context, () => this.#lines(path).slice(0, Math.max(0, options?.maxLines ?? Infinity)));
	}

	async openTextLineReader(path: string, context: Context): Promise<Result<TextLineReader, FileError>> {
		return this.#run(context, () => {
			const text = decode(this.#file(this.#resolve(path)).data);
			const lines: TextLine[] = this.#lines(path).map((line) => ({ text: line, terminated: true }));
			if (!text.endsWith("\n") && lines.length > 0)
				lines[lines.length - 1] = { ...lines.at(-1)!, terminated: false };
			let closed = false;
			return {
				readLine: async (readContext) =>
					this.#run(readContext, () => {
						if (closed) throw fail("invalid", "Text line reader is closed", path);
						return lines.shift();
					}),
				close: async () => {
					closed = true;
				},
			};
		});
	}

	async openBinaryReader(path: string, _options: { noFollow?: boolean } | undefined, context: Context) {
		return this.#run(context, (): BinaryReader => {
			const absolute = this.#resolve(path);
			// Holding the node, like an open file descriptor: a rename does not change what the reader sees.
			const node = this.#file(absolute);
			let closed = false;
			const open = <T>(readContext: Context, fn: () => T) =>
				this.#run(readContext, () => {
					if (closed) throw fail("invalid", "Binary reader is closed", absolute);
					return fn();
				});
			return {
				info: async (readContext) => open(readContext, () => info(absolute, node)),
				read: async (offset, length, readContext) =>
					open(readContext, () => {
						if (![offset, length].every((n) => Number.isSafeInteger(n) && n >= 0)) {
							throw fail("invalid", "Offset and length must be non-negative integers", absolute);
						}
						return node.data.slice(offset, offset + length);
					}),
				scanLines: async ({ startLine, endLine }, readContext) =>
					open(readContext, () => {
						if (
							!Number.isSafeInteger(startLine) ||
							startLine < 0 ||
							(endLine !== undefined && endLine <= startLine)
						) {
							throw fail("invalid", "Invalid line range", absolute);
						}
						const scanner = new LineScanner(startLine, endLine);
						scanner.push(node.data);
						return scanner.finish();
					}),
				close: async () => {
					closed = true;
				},
			};
		});
	}

	async writeFile(path: string, content: string | Uint8Array, context: Context) {
		return this.#run(context, () => this.#write(this.#resolve(path), bytes(content), false));
	}

	async appendFile(path: string, content: string | Uint8Array, context: Context) {
		return this.#run(context, () => this.#write(this.#resolve(path), bytes(content), true));
	}

	async truncateFile(path: string, size: number, context: Context) {
		return this.#run(context, () => {
			const absolute = this.#resolve(path);
			const node = this.#file(absolute);
			const data = new Uint8Array(size);
			data.set(node.data.subarray(0, size));
			Object.assign(node, { data, mtimeMs: Date.now() });

		});
	}

	async flushFile(path: string, context: Context) {
		return this.#run(context, () => void this.#file(this.#resolve(path)));
	}

	async renameFile(sourcePath: string, destinationPath: string, context: Context) {
		return this.#run(context, () => {
			const [from, to] = [this.#resolve(sourcePath), this.#resolve(destinationPath)];
			const node = this.#node(from);
			this.#directory(posix.dirname(to));
			if (isWithin(to, from) && to !== from) throw fail("invalid", "Cannot move a directory into itself", to);
			if (node.kind === "file" && this.volume.nodes.get(to)?.kind === "directory") {
				throw fail("is_directory", "Destination is a directory", to);
			}
			const moved = [...this.volume.nodes].filter(([key]) => isWithin(key, from));
			for (const [key] of [...this.volume.nodes].filter(([key]) => isWithin(key, to) || isWithin(key, from))) {
				this.volume.nodes.delete(key);
			}
			for (const [key, value] of moved) this.volume.nodes.set(to + key.slice(from.length), value);
			this.volume.changed(from, to);
		});
	}

	async fileInfo(path: string, context: Context) {
		return this.#run(context, () => {
			const absolute = this.#resolve(path);
			return info(absolute, this.#node(absolute));
		});
	}

		return this.#run(context, () => this.#children(this.#directory(this.#resolve(path))));
	}

	async openDirReader(path: string, context: Context) {
		return this.#run(context, (): DirReader => {
			const absolute = this.#directory(this.#resolve(path));
			const pending = this.#children(absolute).map((entry) => entry.path);
			let closed = false;
			return {
				next: async (maxEntries, readContext) =>
					this.#run(readContext, () => {
						if (closed || !Number.isSafeInteger(maxEntries) || maxEntries <= 0) {
							throw fail("invalid", closed ? "Directory reader is closed" : "Invalid page size", absolute);

							const entryPath = pending.shift()!;
							const node = this.volume.nodes.get(entryPath);
							if (node !== undefined) entries.push(info(entryPath, node)); // removed meanwhile: skipped
						}
						return { entries, done: pending.length === 0 };
					}),
				close: async () => {
					closed = true;
				},
			};
		});
	}

	async watch(targets: readonly WatchTarget[], onChange: (change: WatchChange) => void, context: Context) {
		return this.#run(context, (): FileWatcher => {
			const watched = targets.map((target) => ({ ...target, path: this.#resolve(target.path) }));

			this.volume.listeners.add(listener);

				mode: "native",
				close: async () => {
					this.volume.listeners.delete(listener);
				},
			};
		});
	}

		return this.#run(context, () => {
			const absolute = this.#resolve(path);
			this.#node(absolute);

		});
	}

	}

		return this.#run(context, () => {
			const absolute = this.#resolve(path);

			if (node !== undefined) break;

		else this.volume.nodes.set(path, { kind: "file", data: next, mtimeMs: Date.now() });
		this.volume.changed(path);
	}
}

	joined.set(b, a.length);

/**
 * Discovers instructions (AGENTS.md and the like) and skills through an `ExecutionEnv`, so it finds what the
 * conversation's tools would find: on this machine, over SSH, in a container, or in memory. It only reads and lists;
 * paths are resolved and joined by the environment, in the environment's own path rules.
 *
 * Results are cached per file namespace and directory (`env.id`, `env.cwd`), so a conversation's requests do not walk
 * the tree again, and dropped when `env.watch()` reports a change: rendered sections stay identical until a file
 * changes, which keeps provider prompt caches warm.
 */
export class ResourceLoader {
	readonly #options: Required<ResourceLoaderOptions>;
	readonly #cache = new Map<string, CacheEntry>();

	constructor(options: ResourceLoaderOptions = {}) {
		this.#options = { ...DEFAULTS, ...options };
	}

	/** What `env` sees from its working directory. Cancelling `context` stops only this wait, not the shared load. */
	load(env: ExecutionEnv, context: Context): Promise<Resources> {
		const key = `${env.id}\0${env.cwd}`;
		let entry = this.#cache.get(key);
		if (entry === undefined) {
			const created = new CacheEntry((self) => this.#discover(env, key, self));
			created.resources.catch(() => this.#drop(key, created));
			this.#cache.set(key, created);
			entry = created;
		}
		return awaitWithContext(entry.resources, context);
	}

	async close(context: Context): Promise<void> {
		const entries = [...this.#cache.values()];
		this.#cache.clear();
		for (const entry of entries) await entry.watcher?.close(context);
	}

	async #discover(env: ExecutionEnv, key: string, entry: CacheEntry): Promise<Resources> {
		// Shared by every request of this namespace and directory, so no single request's cancellation applies.
		const context = BACKGROUND_CONTEXT;
		const { contextFileNames } = this.#options;
		const dirs = await this.#contextDirs(env, context);
		const candidates = await Promise.all(
			dirs.map((dir) => Promise.all(contextFileNames.map(async (name) => join(env, [dir, name], context)))),
		);
		const roots = await Promise.all(this.#options.skillDirs.map(async (dir) => absolute(env, dir, context)));

		// Watch before reading: a change made during the load still drops the entry. Without watching, nothing is cached.
		const watched = await env.watch(
			[
				...candidates.flat().map((path) => ({ path })),
				...roots.map((path) => ({ path, recursive: true, exclude: { hidden: true, names: ["node_modules"] } })),
			],
			() => this.#drop(key, entry),
			context,
		);
		if (!watched.ok || this.#cache.get(key) !== entry) {
			this.#drop(key, entry);
			if (watched.ok) await watched.value.close(context);
		} else {
			entry.watcher = watched.value;
		}

		const [contextFiles, skills] = await Promise.all([
			Promise.all(candidates.map((paths) => firstReadable(env, paths, context))),
			Promise.all(roots.map((root) => skillsIn(env, root, 0, context))),
		]);
		const seen = new Set<string>();
		return {
			contextFiles: contextFiles.filter((file) => file !== undefined),
			skills: skills.flat().filter((skill) => !seen.has(skill.name) && seen.add(skill.name)),
		};
	}

	/** Global directories, then the working directory's ancestors from the root down to it. */
	async #contextDirs(env: ExecutionEnv, context: Context): Promise<string[]> {
		const ancestors: string[] = [];
		for (let dir = await absolute(env, ".", context); ; ) {
			ancestors.unshift(dir);
			const parent = await join(env, [dir, ".."], context);
			if (parent === dir) break;
			dir = parent;
		}
		const globals = await Promise.all(this.#options.globalDirs.map(async (dir) => absolute(env, dir, context)));
		return [...new Set([...globals, ...ancestors])];
	}

	#drop(key: string, entry: CacheEntry): void {
		if (this.#cache.get(key) === entry) this.#cache.delete(key);
		void entry.watcher?.close(BACKGROUND_CONTEXT);
	}
}

 * further; other directories are searched; Markdown files directly in a root are skills when they have a description.
 */
async function skillsIn(env: ExecutionEnv, dir: string, depth: number, context: Context): Promise<Skill[]> {
	const listed = await env.listDir(dir, context);
	if (!listed.ok) return [];
	const entries = await Promise.all(listed.value.map((entry) => followLink(env, entry, context)));
	const skillFile = entries.find((entry) => entry?.name === "SKILL.md" && entry.kind === "file");
	if (skillFile !== undefined) return optional(await readSkill(env, skillFile.path, true, context));
	const nested = await Promise.all(
		entries.map(async (entry): Promise<Skill[]> => {
			if (entry === undefined || entry.name.startsWith(".") || entry.name === "node_modules") return [];
			if (entry.kind === "directory") {
				return depth < MAX_SKILL_DEPTH ? skillsIn(env, entry.path, depth + 1, context) : [];
			}
			return depth === 0 && entry.name.endsWith(".md")
				? optional(await readSkill(env, entry.path, false, context))
				: [];
		}),
	);
	return nested.flat();
}

		return {};
	}
}

		options?: AuthOperationOptions,
	): Promise<Credential | undefined> {

			if (next === undefined) return all[providerId];
			all[providerId] = next;

			return next;
		}, options);

		return this.#locked(async (all) => {
			delete all[providerId];
			await this.#save(all);
		}, options);
	}

		try {
			return JSON.parse(await readFile(this.path, "utf8")) as Credentials;
		} catch (error) {
			if ((error as NodeJS.ErrnoException).code === "ENOENT") return {};
			throw error;
		}
	}

	async #save(all: Credentials): Promise<void> {
		await writeFile(this.path, JSON.stringify(all, null, 2), { encoding: "utf8", mode: 0o600 });
	}

	async #locked<T>(change: (all: Credentials) => Promise<T>, options?: AuthOperationOptions): Promise<T> {
		options?.signal?.throwIfAborted();
		// The lock needs the file to exist.
		await mkdir(dirname(this.path), { recursive: true, mode: 0o700 });
		await writeFile(this.path, "{}", { flag: "wx", mode: 0o600 }).catch((error: NodeJS.ErrnoException) => {
			if (error.code !== "EEXIST") throw error;
		});
		const release = await lockfile.lock(this.path, {
			realpath: false,
			stale: 30_000,
			retries: { retries: 30, minTimeout: 10, maxTimeout: 1000 },
		});

			return await change(await this.#load());
		} finally {

	}

async function run(modelSpec: string, prompt: string): Promise<void> {
	const models = createModels(values.auth);
	const model = parseModelRef(models, modelSpec);
	if (values.session !== undefined) await mkdir(dirname(values.session), { recursive: true });
	const storage = values.session === undefined ? new MemoryStorage() : await openNodeSqliteStorage(values.session);
	const { harness, close } = await openHarness(
		{ storage, models, envs: await environments(), onReport: (error) => console.error("[report]", error) },
		context,
	);
	try {
		// Continue whatever the last process left unfinished in this session.
		harness.resume();
		const root = await harness.root(context);
		const cwd = defaultCwd();
		await root.configure({ model, ...(cwd === undefined ? {} : { cwd }) }, context);
		if (values.env !== undefined) {
			const uri = values.env;
			await root.commit(async (tx) => {
				(await tx.doc(EnvDoc, root.id)).uri = uri;
			}, context);
		}

		const events = await watchEvents(harness, root.id, context);
		events.start(async (batch) => batch.forEach(print));
		const settled = await (await root.submit({ type: "input", content: prompt }, context)).wait(context);
		await events.stop();
		if (settled.status === "unanswered") {
			console.error(`\nunanswered: ${settled.reason}`);
			process.exitCode = 1;
		}
		const usage = await harness.usage(context);
		const cost = Object.values(usage.models).reduce((sum, entry) => sum + entry.cost.total, 0);
		console.error(`\n[${modelSpec}, $${cost.toFixed(4)}]`);
	} finally {
		await close(context);
	}
}
